# tread cp 环境复制 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `tread cp <src> <dst>` 把一个环境的工具链复制成新环境，复制完两者彻底无关。

**Architecture:** 新增 `src/copy.ts`，遍历 src 逐字节拷真实文件、跳过 volatile 清单、跳过指向真 home 的 symlink（由 `ensureSkeleton(dst)` 按 dst 自己的配置重建）、把其它 symlink 解引用成真实文件；落地到临时目录后 rename，再重写内容里残留的 src 绝对路径。输出复用 `statusOne`。

**Tech Stack:** Bun + TypeScript，`bun test`（bun:test），无新依赖。

**Spec:** `docs/superpowers/specs/2026-08-08-env-copy-design.md`

## Global Constraints

- volatile 清单按 **env 相对路径精确匹配**，不能按目录名匹配：`.claude/cache` 要排，`.claude/plugins/cache` 是插件本体必须留。
- dst 里不允许残留任何指向 src 的路径——不管是 symlink 还是文件内容里的绝对路径。
- 只处理常规文件、目录、symlink；socket / fifo / 设备节点跳过。
- 代码注释用英文（跟现有源码一致），测试的 `test()` 描述用中文（跟现有测试一致）。
- 不加新依赖，不改既有命令行为。
- 每个 task 结束时 `bun test` 与 `bunx tsc --noEmit` 都必须过。

- [ ] **Step 4: 写 `src/copy.ts`**

```ts
import fs from "node:fs";
import path from "node:path";
import { AGENTS, AGENT_SPECS } from "./agents.ts";
import { ensureSkeleton, requireEnv } from "./env.ts";
import { envDir, envsDir, realHome, validateEnvName } from "./paths.ts";

/** How far a chain of dereferenced symlinks may go before we give up. */
const MAX_DEPTH = 8;

/**
 * Env-relative paths a copy leaves behind, beyond the per-agent lists.
 *
 * Exact paths, not names: matching by name would take `.claude/plugins/cache`
 * — the plugin bodies, a third of the environment — along with the throwaway
 * `.claude/cache`, and the status table would still show the plugins because
 * it counts entries in installed_plugins.json.
 */
const SHARED_VOLATILE = [
  ".tread/sync.json",              // regenerated for dst; src's is src's ledger
  ".local/state",                  // tread's own state, plus gh/claude lock files
  "Library/Caches",                // cursor's compile cache
  "Library/Application Support",   // cursor desktop's skill index db, clawhub state
];

export function volatilePaths(): string[] {
  const out = [...SHARED_VOLATILE];
  for (const a of AGENTS) {
    for (const rel of AGENT_SPECS[a].volatile) {
      out.push(`${AGENT_SPECS[a].dir}/${rel}`);
    }
  }
  return out;
}

export interface CopyResult {
  root: string;
  files: number;
  rewritten: number;
  /** env-relative paths left behind: odd file types, too deep, too big */
  skipped: string[];
}

function intoRealHome(target: string): boolean {
  return target === realHome() || target.startsWith(realHome() + path.sep);
}

/**
 * Copy an environment's own content into `dst`.
 *
 * Links into the real home are left out entirely: `ensureSkeleton(dst)` puts
 * them back from dst's own configuration, which is the current answer to what
 * is shared — src's links are a snapshot of whatever the config said the day
 * it was created. Any other symlink is dereferenced instead of recreated,
 * because a link is how dst would stay tied to src.
 */
function copyTree(src: string, dst: string, result: CopyResult): void {
  const volatile = new Set(volatilePaths());

  const copyFile = (from: string, to: string): void => {
    fs.copyFileSync(from, to);
    fs.chmodSync(to, fs.statSync(from).mode & 0o7777);
    result.files++;
  };

  const walk = (rel: string, depth: number): void => {
    const dir = rel ? path.join(src, rel) : src;
    for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
      const childRel = rel ? `${rel}/${entry.name}` : entry.name;
      if (volatile.has(childRel)) continue;
      const from = path.join(src, childRel);
      const to = path.join(dst, childRel);

      if (entry.isSymbolicLink()) {
        const target = path.resolve(path.dirname(from), fs.readlinkSync(from));
        if (intoRealHome(target)) continue; // ensureSkeleton rebuilds these
        deref(target, to, childRel, depth);
        continue;
      }
      if (entry.isDirectory()) {
        fs.mkdirSync(to, { recursive: true });
        walk(childRel, depth);
        continue;
      }
      if (entry.isFile()) {
        copyFile(from, to);
        continue;
      }
      result.skipped.push(childRel); // socket, fifo, device node
    }
  };

  /** Copy what a symlink pointed at, so dst holds content rather than a tie to src. */
  const deref = (target: string, to: string, rel: string, depth: number): void => {
    if (depth >= MAX_DEPTH) {
      result.skipped.push(rel);
      return;
    }
    let st: fs.Stats;
    try {
      st = fs.statSync(target); // follows the chain; throws on a broken link
    } catch {
      result.skipped.push(rel);
      return;
    }
    if (st.isFile()) {
      fs.mkdirSync(path.dirname(to), { recursive: true });
      copyFile(target, to);
      return;
    }
    if (!st.isDirectory()) {
      result.skipped.push(rel);
      return;
    }
    fs.mkdirSync(to, { recursive: true });
    for (const entry of fs.readdirSync(target, { withFileTypes: true })) {
      const childTarget = path.join(target, entry.name);
      const childTo = path.join(to, entry.name);
      const childRel = `${rel}/${entry.name}`;
      if (entry.isSymbolicLink()) {
        const next = path.resolve(target, fs.readlinkSync(childTarget));
        if (intoRealHome(next)) continue;
        deref(next, childTo, childRel, depth + 1);
      } else if (entry.isDirectory()) {
        deref(childTarget, childTo, childRel, depth + 1);
      } else if (entry.isFile()) {
        copyFile(childTarget, childTo);
      } else {
        result.skipped.push(childRel);
      }
    }
  };

  walk("", 0);
}

export function copyEnv(srcName: string, dstName: string): CopyResult {
  const src = requireEnv(srcName);
  const dst = envDir(dstName); // validates the name
  if (fs.existsSync(dst)) {
    throw new Error(
      `"${dstName}" already exists\n\n  ${dst}\n  tread rm ${dstName}   to replace it`,
    );
  }
  if (path.resolve(src) === path.resolve(dst)) {
    throw new Error(`cannot copy "${srcName}" onto itself`);
  }

  // land in a temp directory and rename: the long part is the byte copy, and
  // a half-copied environment is worse than none — it shows up in `tread ls`
  // and its status table looks plausible
  const staging = path.join(envsDir(), `.cp-${dstName}.${process.pid}`);
  fs.rmSync(staging, { recursive: true, force: true });
  const result: CopyResult = { root: dst, files: 0, rewritten: 0, skipped: [] };
  try {
    fs.mkdirSync(staging, { recursive: true });
    copyTree(src, staging, result);
    fs.renameSync(staging, dst);
  } catch (e) {
    fs.rmSync(staging, { recursive: true, force: true });
    throw e;
  }

  // after the rename, so the links and the generated tread skill are written
  // against the final path rather than the staging one
  ensureSkeleton(dst);
  return result;
}
```

函数：

```ts
/**
 * Replace every mention of `from` with `to` in the copied files.
 *
 * A byte-identical copy is not an independent environment: agents bake the
 * absolute config path into what they write — claude's hook commands and
 * installed_plugins.json, cursor's hooks.json, kimi's extra_skill_dirs, and
 * whatever a skill's own installer computed. Left alone, dst would read src's
 * directories and say nothing about it.
 *
 * Literal, whole-path replacement only. The env *name* is never substituted:
 * a name like "test" would hit prose everywhere.
 */
export function rewritePaths(
  root: string,
  from: string,
  to: string,
): { rewritten: number; skipped: string[] } {
  let rewritten = 0;
  const skipped: string[] = [];

  const walk = (rel: string): void => {
    const dir = rel ? path.join(root, rel) : root;
    for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
      const childRel = rel ? `${rel}/${entry.name}` : entry.name;
      const p = path.join(root, childRel);
      // never follow a link: they point into the real home, which is not ours
      if (entry.isSymbolicLink()) continue;
      if (entry.isDirectory()) {
        walk(childRel);
        continue;
      }
      if (!entry.isFile()) continue;
      let st: fs.Stats;
      try {
        st = fs.statSync(p);
      } catch {
        continue;
      }
      if (st.size > MAX_REWRITE_BYTES) {
        skipped.push(childRel);
        continue;
      }
      const buf = fs.readFileSync(p);
      // a NUL in the first 8 KB means binary; substituting into it would
      // corrupt it, and no agent config is binary
      if (buf.subarray(0, 8192).includes(0)) continue;
      const text = buf.toString("utf8");
      if (!text.includes(from)) continue;
      fs.writeFileSync(p, text.split(from).join(to), { mode: st.mode & 0o7777 });
      rewritten++;
    }
  };

  walk("");
  return { rewritten, skipped };
}
```

`copyEnv` 结尾改成：

```ts
  ensureSkeleton(dst);
  const rw = rewritePaths(dst, src, dst);
  result.rewritten = rw.rewritten;
  result.skipped.push(...rw.skipped);
  return result;
```

function cpCommand(args: string[], out: Out): number {
  const [src, dst] = args;
  // both names are always explicit: `cp` takes two arguments everywhere else,
  // and a one-argument form reads as source or target depending on whether an
  // environment happens to be active
  if (!src || !dst) throw new Error("cp needs two names\n\n  tread cp <src> <dst>");
  const c = color(colorsEnabled());
  const r = copyEnv(src, dst);
  const plural = (n: number, word: string) => `${n} ${word}${n === 1 ? "" : "s"}`;
  const notes = [
    "skipped sessions and caches",
    `${plural(r.rewritten, "path")} rewritten`,
  ];
  if (r.skipped.length > 0) notes.push(`${plural(r.skipped.length, "file")} skipped`);
  out(`copied  ${c.bold(src)} → ${c.brightGreen(dst)}\n\n`);
  out(statusOne(r.root, dst, false));
  // the only place the copy admits it was not byte-for-byte
  out(c.dim(notes.join(" · ")) + "\n");
  return 0;
}
